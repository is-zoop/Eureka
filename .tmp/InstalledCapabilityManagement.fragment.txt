function InstalledCapabilityManagement({ item, scope, busy, onOpenDetails, onAction, onSetDisabled }: {
  item: ListedCapability; scope: InstallScope; busy: boolean;
  onOpenDetails: (item: OrganizationExtension) => void;
  onAction: (item: ListedCapability, action: InstalledAction) => void;
  onSetDisabled: (item: OrganizationExtension, disabled: boolean) => void;
}) {
  const { t, locale } = useI18n();
  const canUpdate = item.type !== "MCP" && !item.marketUnavailable;
  const formatDate = (value?: string | null) => {
    if (!value) return "—";
    const date = new Date(value);
    return Number.isNaN(date.getTime()) ? value : date.toLocaleString(locale);
  };
  const fields = [
    [t("extensions.capabilityType"), t(item.type === "Skill" ? "extensions.skills" : item.type === "Plugin" ? "extensions.plugins" : "extensions.mcp")],
    [t("extensions.installScope"), t(scope === "global" ? "extensions.global" : "extensions.project")],
    [t("extensions.version"), item.version ? `v${item.version}` : "—"],
    ...(item.marketVersion ? [[t("extensions.latestVersion"), `v${item.marketVersion}`]] : []),
    [t("extensions.developer"), item.author || "—"],
    [t("extensions.department"), item.department || "—"],
    [t("extensions.installedAt"), formatDate(item.installedAt)],
    [t("extensions.updatedAt"), formatDate(item.updatedAt)],
    ...(item.type === "MCP" ? [
      [t("extensions.connectionType"), item.connectType || "—"],
      [t("extensions.lifecycle"), item.lifecycle === "eager" ? "Eager" : item.lifecycle === "keep-alive" ? "Keep alive" : "Lazy"],
      [t("extensions.idleTimeout"), String(item.idleTimeout ?? 10)],
      ...(item.serverUrl ? [[t("extensions.serverUrl"), item.serverUrl]] : []),
    ] : []),
  ];
  return <article data-installed-management className="w-full max-w-5xl px-1 py-2 text-[var(--text)] sm:px-4 sm:py-4">
    <div className="flex flex-wrap items-start justify-between gap-5">
      <div className="flex min-w-0 flex-1 items-start gap-4">
        <CapabilityIcon item={item} />
        <div className="min-w-0 flex-1">
          <h2 className="break-words text-xl font-semibold">{item.name}</h2>
          <div className="mt-2 flex flex-wrap items-center gap-2 text-xs text-[var(--text-muted)]">
            <span className="rounded-[var(--radius-control)] bg-[var(--bg-hover)] px-2 py-1">{item.type}</span>
            {item.plugin && <span>{t(`extensions.pluginState.${item.plugin.state}`)}</span>}
            {hasSkillUpdate(item) && <span className="font-medium text-[var(--accent)]">{t("extensions.updateAvailable")}</span>}
          </div>
        </div>
      </div>
      <label className="inline-flex shrink-0 cursor-pointer items-center gap-3 py-1 text-sm">
        <span>{t(item.disabled ? "extensions.pluginState.disabled" : "extensions.managementEnabled")}</span>
        <Switch checked={!item.disabled} disabled={busy} aria-label={`${item.disabled ? t("extensions.enable") : t("extensions.disable")}${item.name}`} onCheckedChange={checked => onSetDisabled(item, !checked)} />
      </label>
    </div>
    <p className="mt-5 whitespace-pre-wrap break-words text-sm leading-6 text-[var(--text-muted)]">{item.marketUnavailable ? t("extensions.marketUnavailable") : item.description || t("extensions.marketNoDescription")}</p>
    <section className="mt-6 border-y border-[var(--border)] py-5" aria-label={t("extensions.manage")}>
      <div className="flex flex-wrap items-center gap-2">
        {canUpdate && <Button variant="outline" size="sm" disabled={busy} onClick={() => onAction(item, "update")}><RefreshCwIcon size={14} className="mr-1.5" aria-hidden="true" />{t("extensions.update")}</Button>}
        {item.type === "MCP" && <Button variant="outline" size="sm" disabled={busy} onClick={() => onAction(item, "configure")}><SlidersHorizontalIcon size={14} className="mr-1.5" aria-hidden="true" />{t("extensions.configure")}</Button>}
        <Button variant="outline" size="sm" disabled={busy} onClick={() => onOpenDetails(item)}><EyeIcon size={14} className="mr-1.5" aria-hidden="true" />{t("extensions.viewDetails")}</Button>
        <Button variant="destructive" size="sm" className="sm:ml-auto" disabled={busy} onClick={() => onAction(item, "uninstall")}><Trash2Icon size={14} className="mr-1.5" aria-hidden="true" />{t("extensions.uninstall")}</Button>
      </div>
      <p className="mt-3 text-xs leading-5 text-[var(--text-muted)]">{t("extensions.managementReloadHint")}</p>
    </section>
    <section className="pt-6" aria-label={t("extensions.basicInformation")}>
      <h3 className="text-sm font-semibold">{t("extensions.basicInformation")}</h3>
      <dl className="mt-5 grid grid-cols-1 gap-x-8 gap-y-6 min-[480px]:grid-cols-2 lg:grid-cols-3">
        {fields.map(([label, value]) => <div key={label} className="min-w-0"><dt className="text-xs text-[var(--text-muted)]">{label}</dt><dd className="mt-2 break-words text-sm">{value}</dd></div>)}
        {item.tags.length > 0 && <div className="min-w-0 min-[480px]:col-span-2 lg:col-span-3"><dt className="text-xs text-[var(--text-muted)]">{t("extensions.capabilityTags")}</dt><dd className="mt-2 flex flex-wrap gap-2">{item.tags.map(tag => <span key={tag} className="rounded-[var(--radius-control)] bg-[var(--bg-hover)] px-2 py-1 text-xs text-[var(--text-muted)]">{tag}</span>)}</dd></div>}
      </dl>
    </section>
  </article>;
}
